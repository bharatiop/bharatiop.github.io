\cvsection{Research Interests}
\cventry{Nuclear\\ Structure}{%
  Ground-state properties of finite nuclei; nuclear energy-density functionals;
  relativistic mean-field and Skyrme Hartree--Fock models; development of
  effective interactions (IOPB-I, G3, NITR, NITR-I); parity doublets and shape
  coexistence; multifragmentation and fission yields; density dependence of the
  symmetry energy; neutron skins and the PREX/CREX constraints; equation of
  state of dense and warm nuclear matter.}

\cventry{Compact-Star\\ Astrophysics}{%
  Static, rotating and anisotropic neutron-star structure --- mass, radius,
  moment of inertia, quadrupole deformation, tidal deformability; radial and
  non-radial ($f$-, $p$-, $g$-mode) oscillations; proto-neutron stars at finite
  entropy; exotic core composition (hyperons, $\Delta$ baryons, quarkyonic and
  quark matter); dark-matter-admixed compact objects and two-fluid models;
  spacetime curvature and compact stars in modified gravity;
  equation-of-state-insensitive (universal) relations.}

\cventry{Computational\\ Methods}{%
  Bayesian inference and Markov-chain Monte Carlo; nonparametric
  equation-of-state representations; machine-learning-assisted parameter
  estimation; large-scale numerical solution of the TOV and stellar-perturbation
  equations.}
